	chain DOCKER_OUTPUT {
		ip daddr 127.0.0.11 tcp dport 53 counter packets 0 bytes 0 xt target "DNAT"
		ip daddr 127.0.0.11 udp dport 53 counter packets 0 bytes 0 xt target "DNAT"
	}

	chain OUTPUT {
		type nat hook output priority dstnat; policy accept;
		ip daddr 127.0.0.11 counter packets 0 bytes 0 jump DOCKER_OUTPUT
	}

	chain DOCKER_POSTROUTING {
		ip saddr 127.0.0.11 tcp sport 33935 counter packets 0 bytes 0 xt target "SNAT"
		ip saddr 127.0.0.11 udp sport 39991 counter packets 0 bytes 0 xt target "SNAT"
	}

	chain POSTROUTING {
		type nat hook postrouting priority srcnat; policy accept;
		ip daddr 127.0.0.11 counter packets 0 bytes 0 jump DOCKER_POSTROUTING
	}
}

	set blocked_v4 {
		type ipv4_addr
		timeout 1m
	}

	chain forward_guard {
		type filter hook forward priority filter - 10; policy accept;
		ip saddr @blocked_v4 ip daddr 10.20.20.0/24 drop
	}
}
